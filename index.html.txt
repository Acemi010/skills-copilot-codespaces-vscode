      <input type="text" id="username" placeholder="Adınızı girin..." />

      <button onclick="sendMessage()">Gönder</button>

  <!-- JavaScript -->
  <script>
    // Sohbet penceresini güncelleyen fonksiyon
    function updateChatBox(message, sender) {
      const chatBox = document.getElementById('chatBox');
      const newMessage = document.createElement('div');
      
      // Mesaj türüne göre sağa veya sola hizalama
      newMessage.classList.add(sender === 'user' ? 'user-message' : 'bot-message');
      
      newMessage.textContent = message;
      chatBox.appendChild(newMessage);

      // Sohbeti en son mesaja kaydır
      chatBox.scrollTop = chatBox.scrollHeight;
    }

    // Mesaj gönderme fonksiyonu
    function sendMessage() {
      const usernameInput = document.getElementById('username');
      const messageInput = document.getElementById('message');

      const username = usernameInput.value.trim();
      const message = messageInput.value.trim();

      // Kullanıcı adı ve mesaj kontrolü
      if (!username || !message) {
        alert('Lütfen kullanıcı adı ve mesaj girin!');
        return;
      }

      // Kullanıcının mesajını ekle
      updateChatBox(`${username}: ${message}`, 'user');

      // Botun yanıtını simüle et (basit bir yanıt)
      setTimeout(() => {
        updateChatBox(`Bot: ${generateBotResponse()}`, 'bot');
      }, 1000);

      // Mesajı temizle
      messageInput.value = '';
    }

    // Bot için basit yanıt generator fonksiyonu
    function generateBotResponse() {
      const responses = [
        "Merhaba! Size nasıl yardımcı olabilirim?",
        "Bugün hava çok güzel, değil mi?",
        "Bu sohbet çok eğlenceli!",
        "Bunun hakkında daha fazla bilgi verebilir miyim?",
        "Nasıl yardımcı olabilirim?"
      ];
      
      return responses[Math.floor(Math.random() * responses.length)];
    }
  </script>
